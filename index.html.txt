    function buildGrid(cs) {
      const g = Array.from({length:ROWS},()=>Array(COLS).fill(undefined));
      for (const c of cs) if (!c.removing && c.row>=0 && c.row<ROWS) g[c.row][c.col]=c;
      return g;
    }
    function getZone(candy, all) {
      if (candy.special==="striped-h") return all.filter(c=>c.row===candy.row&&!c.removing).map(c=>c.id);
      if (candy.special==="striped-v") return all.filter(c=>c.col===candy.col&&!c.removing).map(c=>c.id);
      if (candy.special==="bomb")      return all.filter(c=>Math.abs(c.row-candy.row)<=1&&Math.abs(c.col-candy.col)<=1&&!c.removing).map(c=>c.id);
      return [];
    }
    function applyRun(run, dir, rem, tx) {
      const hasS = run.some(c=>c.special!=="none");
      if (run.length>=5&&!hasS) { const mid=run[Math.floor(run.length/2)]; run.forEach(c=>rem.add(c.id)); rem.delete(mid.id); tx.set(mid.id,"bomb"); }
      else if (run.length===4&&!hasS) { const mid=run[1]; run.forEach(c=>rem.add(c.id)); rem.delete(mid.id); tx.set(mid.id,dir); }
      else run.forEach(c=>rem.add(c.id));
    }
    function findMatches(cs, extra=new Set()) {
      const rem=new Set(extra), tx=new Map();
      const grid=buildGrid(cs);
      for (let r=0;r<ROWS;r++){let c=0;while(c<COLS){const cell=grid[r][c];if(!cell){c++;continue;}let len=1;while(c+len<COLS&&grid[r][c+len]?.type===cell.type)len++;if(len>=3)applyRun(Array.from({length:len},(_,i)=>grid[r][c+i]),"striped-h",rem,tx);c+=len;}}
      for (let col=0;col<COLS;col++){let r=0;while(r<ROWS){const cell=grid[r][col];if(!cell){r++;continue;}let len=1;while(r+len<ROWS&&grid[r+len]?.[col]?.type===cell.type)len++;if(len>=3)applyRun(Array.from({length:len},(_,i)=>grid[r+i][col]),"striped-v",rem,tx);r+=len;}}
      let changed=true;
      while(changed){changed=false;for(const id of [...rem]){if(tx.has(id))continue;const c=cs.find(x=>x.id===id);if(!c||c.special==="none")continue;for(const zid of getZone(c,cs))if(!rem.has(zid)){rem.add(zid);changed=true;}}}
      for(const id of tx.keys())rem.delete(id);
      return {toRemove:rem,toTransform:tx};
    }
    function applyGravity(remaining) {
      let res=remaining;
      for(let col=0;col<COLS;col++){
        const col_=res.filter(c=>c.col===col).sort((a,b)=>a.row-b.row);
        const empty=ROWS-col_.length;
        const filled=Array.from({length:empty},(_,i)=>mkCandy(-empty+i,col));
        const all=[...filled,...col_].map((c,i)=>({...c,row:i}));
        res=[...res.filter(c=>c.col!==col),...all];
      }
      return res;
    }

    function Game() {
      const [candies, setCandies] = useState(()=>initBoard());
      const candiesRef = useRef(candies);
      const [score,    setScore]    = useState(0);
      const [moves,    setMoves]    = useState(MAX_MOVES);
      const [best,     setBest]     = useState(getBest);
      const [gameOver, setGameOver] = useState(false);
      const [newBest,  setNewBest]  = useState(false);
      const [selected, setSelected] = useState(null);
      const busyRef  = useRef(false);
      const scoreRef = useRef(0);
      const movesRef = useRef(MAX_MOVES);
      const initRef  = useRef(false);
      const offset   = Math.floor((CELL-GEM)/2);

      const update = useCallback(next=>{candiesRef.current=next;setCandies(next);},[]);

      const endGame = useCallback(fs=>{
        const prev=getBest(),nb=fs>prev;
        if(nb){saveBest(fs);setBest(fs);}
        setNewBest(nb);setGameOver(true);
      },[]);

      const processMatches = useCallback(
        (cur,acc,extra=new Set())=>new Promise(resolve=>{
          const {toRemove,toTransform}=findMatches(cur,extra);
          if(!toRemove.size&&!toTransform.size){
            scoreRef.current=acc;setScore(acc);
            if(movesRef.current<=0)endGame(acc);
            resolve();return;
          }playMatchSound(toRemove.size);
          const newAcc=acc+toRemove.size*10+toTransform.size*20;
          const marked=cur.map(c=>{
            if(toTransform.has(c.id))return{...c,special:toTransform.get(c.id)};
            if(toRemove.has(c.id))return{...c,removing:true};
            return c;
          });
          update(marked);
          setTimeout(()=>{
            const rem=applyGravity(marked.filter(c=>!c.removing));
            update(rem);
            setTimeout(()=>processMatches(rem,newAcc).then(resolve),T_GRAVITY);
          },T_REMOVE);
        }),
        [endGame,update]
      );

      useEffect(()=>{
        if(initRef.current)return;initRef.current=true;busyRef.current=true;
        setTimeout(()=>processMatches(candiesRef.current,0).then(()=>{busyRef.current=false;}),T_INIT);
      },[processMatches]);

      const swap = useCallback((idA,idB)=>{
        if(busyRef.current||movesRef.current<=0)return;
        busyRef.current=true;setSelected(null);
        movesRef.current--;setMoves(movesRef.current);
        const cur=candiesRef.current;
        const a=cur.find(c=>c.id===idA),b=cur.find(c=>c.id===idB);
        const sw=cur.map(c=>c.id===idA?{...c,row:b.row,col:b.col}:c.id===idB?{...c,row:a.row,col:a.col}:c);
        update(sw);
        const extra=new Set();
        for(const sc of [a,b].filter(c=>c.special!=="none")){extra.add(sc.id);for(const z of getZone(sc,sw))extra.add(z);}
        setTimeout(()=>processMatches(sw,scoreRef.current,extra).then(()=>{busyRef.current=false;}),T_SWAP);
      },[processMatches,update]);

      const handleDown = useCallback(id=>{
        if(busyRef.current||gameOver)return;
        setSelected(prev=>{
          if(prev===null)return id;
          if(prev===id)return null;
          const a=candiesRef.current.find(c=>c.id===prev),b=candiesRef.current.find(c=>c.id===id);
          if(!a||!b)return null;
          if(Math.abs(a.row-b.row)+Math.abs(a.col-b.col)===1){swap(prev,id);return null;}
          return id;
        });
      },[swap,gameOver]);

      const handleEnter = useCallback(id=>{
        if(busyRef.current||gameOver)return;
        setSelected(prev=>{
          if(prev===null||prev===id)return prev;
          const a=candiesRef.current.find(c=>c.id===prev),b=candiesRef.current.find(c=>c.id===id);
          if(!a||!b)return prev;
          if(Math.abs(a.row-b.row)+Math.abs(a.col-b.col)===1){swap(prev,id);return null;}
          return prev;
        });
      },[swap,gameOver]);

      const restart = useCallback(()=>{
        scoreRef.current=0;movesRef.current=MAX_MOVES;busyRef.current=true;
        setScore(0);setMoves(MAX_MOVES);setSelected(null);setGameOver(false);setNewBest(false);
        const fresh=initBoard();update(fresh);
        setTimeout(()=>processMatches(fresh,0).then(()=>{busyRef.current=false;}),T_INIT);
      },[processMatches,update]);

      const gridPx  = COLS*CELL;
      const movePct = moves/MAX_MOVES;
      const barCol  = movePct>0.5?"#00e676":movePct>0.25?"#ffca28":"#ff1744";

      return (
        <div style={{minHeight:"100vh",background:"radial-gradient(ellipse at 50% 0%,#1c0638 0%,#06000f 70%)",display:"flex",flexDirection:"column",alignItems:"center",justifyContent:"center",fontFamily:"'Segoe UI',sans-serif",color:"white",userSelect:"none",position:"relative"}}>
          <GemDefs/>
          <div style={{position:"absolute",width:500,height:500,background:"radial-gradient(circle,rgba(110,40,200,0.2) 0%,transparent 70%)",borderRadius:"50%",pointerEvents:"none"}}/>

          {/* HUD */}
          <div style={{display:"flex",gap:10,marginBottom:12,position:"relative",zIndex:2}}>
            {[
              {label:"Score",value:score,color:"#ff3d9a",glow:"rgba(255,60,154,0.35)"},
              {label:"Moves",value:moves,color:barCol,glow:`${barCol}55`},
              {label:"Best", value:best, color:"#ffd700",glow:"rgba(255,215,0,0.28)"},
            ].map(({label,value,color,glow})=>(
              <div key={label} style={{background:"rgba(255,255,255,0.055)",backdropFilter:"blur(14px)",borderRadius:14,border:`1.5px solid ${color}40`,padding:"9px 22px",textAlign:"center",boxShadow:`0 0 18px ${glow},inset 0 1px 0 rgba(255,255,255,0.08)`}}>
                <div style={{fontSize:9,letterSpacing:1.5,textTransform:"uppercase",opacity:0.55,marginBottom:2}}>{label}</div>
                <div style={{fontSize:23,fontWeight:800,color,lineHeight:1}}>{value}</div>
              </div>
            ))}
          </div>

          {/* Moves bar */}
          <div style={{width:gridPx,height:5,background:"rgba(255,255,255,0.07)",borderRadius:4,marginBottom:9,overflow:"hidden",position:"relative",zIndex:2}}>
            <div style={{width:`${movePct*100}%`,height:"100%",background:`linear-gradient(90deg,${barCol}99,${barCol})`,borderRadius:4,transition:"width 0.3s,background 0.3s",boxShadow:`0 0 8px ${barCol}`}}/>
          </div>

          {/* Legend */}
          <div style={{display:"flex",gap:14,marginBottom:8,fontSize:10,opacity:0.42,position:"relative",zIndex:2}}>
            <span>? Match 4 ? Stripe</span>
            <span>?? Match 5 ? Bomb</span>
            <span>? Swap to activate</span>
          </div>

          {/* Gem key */}
          <div style={{display:"flex",gap:4,marginBottom:8,alignItems:"center",position:"relative",zIndex:2}}>
            {["Ruby","Sapphire","Emerald","Topaz","Amethyst"].map((name,i)=>(
              <div key={i} style={{display:"flex",flexDirection:"column",alignItems:"center",gap:2}}>
                <CandyIcon type={i} special="none"/>
                <span style={{fontSize:8,opacity:0.45}}>{name}</span>
              </div>
            ))}
          </div>

          {/* Grid */}
          <div style={{position:"relative",width:gridPx,height:ROWS*CELL,background:"linear-gradient(155deg,#110328 0%,#08011a 100%)",borderRadius:14,border:"1.5px solid rgba(130,60,200,0.3)",boxShadow:"0 24px 70px rgba(0,0,0,0.75),inset 0 0 40px rgba(70,15,140,0.15)",overflow:"hidden",touchAction:"none",filter:gameOver?"brightness(0.28) blur(4px)":"none",transition:"filter 0.4s",zIndex:2}}
            onPointerLeave={()=>{if(!busyRef.current)setSelected(null);}}>
            <svg width={gridPx} height={ROWS*CELL} style={{position:"absolute",inset:0,pointerEvents:"none",opacity:0.1}}>
              <defs><pattern id="cg" width={CELL} height={CELL} patternUnits="userSpaceOnUse"><path d={`M ${CELL} 0 L 0 0 0 ${CELL}`} fill="none" stroke="rgba(255,255,255,0.8)" strokeWidth="0.5"/></pattern></defs>
              <rect width="100%" height="100%" fill="url(#cg)"/>
            </svg>
            {candies.map(candy=>(
              <GemCell key={candy.id} candy={candy} selected={selected===candy.id} onDown={handleDown} onEnter={handleEnter} offset={offset}/>
            ))}
          </div>

          <div style={{marginTop:10,opacity:0.3,fontSize:10.5,position:"relative",zIndex:2}}>
            Click or drag to swap adjacent gems ó match 3+ to score!
          </div>
          <div style={{marginTop:8,fontSize:20,fontWeight:"bold",color:"#ff3d9a",letterSpacing:1,direction:"rtl",textShadow:"0 0 16px rgba(255,60,154,0.8),0 0 32px rgba(255,60,154,0.3)",position:"relative",zIndex:2}}>
            «··⁄»Â „·ﬂ · ‰Ê›Ì— ÊÕÌœ ”„Ì—
          </div>

          {/* Game Over */}
          {gameOver&&(
            <div style={{position:"absolute",inset:0,zIndex:10,display:"flex",alignItems:"center",justifyContent:"center",animation:"fadeIn 0.35s ease"}}>
              <div style={{background:"linear-gradient(135deg,rgba(16,3,40,0.97),rgba(32,8,68,0.97))",border:"1.5px solid rgba(255,60,154,0.5)",borderRadius:22,padding:"42px 50px",textAlign:"center",boxShadow:"0 0 60px rgba(255,60,154,0.28),0 24px 80px rgba(0,0,0,0.9)",backdropFilter:"blur(22px)",minWidth:290}}>
                {newBest&&<div style={{fontSize:11,letterSpacing:2,color:"#ffd700",marginBottom:10,textTransform:"uppercase"}}>? New Best Score! ?</div>}
                <div style={{fontSize:34,fontWeight:800,marginBottom:4}}>Game Over</div>
                <div style={{opacity:0.45,fontSize:14,marginBottom:30}}>No moves remaining</div>
                <div style={{display:"flex",gap:28,justifyContent:"center",marginBottom:34}}>
                  {[{label:"Your Score",value:score,color:"#ff3d9a"},{label:"Best Score",value:best,color:"#ffd700"}].map(({label,value,color},i,a)=>(
                    <Fragment key={label}>
                      <div>
                        <div style={{fontSize:11,opacity:0.5,letterSpacing:1.5,textTransform:"uppercase",marginBottom:6}}>{label}</div>
                        <div style={{fontSize:40,fontWeight:800,color}}>{value}</div>
                      </div>
                      {i<a.length-1&&<div style={{width:1,background:"rgba(255,255,255,0.07)"}}/>}
                    </Fragment>
                  ))}
                </div>
                <button onClick={restart}
                  style={{background:"linear-gradient(135deg,#ff3d9a,#c200f0)",border:"none",borderRadius:50,padding:"14px 50px",color:"white",fontSize:17,fontWeight:700,cursor:"pointer",boxShadow:"0 4px 22px rgba(255,60,154,0.55)",transition:"transform 0.13s,box-shadow 0.13s"}}
                  onMouseEnter={e=>{e.target.style.transform="scale(1.06)";e.target.style.boxShadow="0 6px 30px rgba(255,60,154,0.75)";}}
                  onMouseLeave={e=>{e.target.style.transform="scale(1)";e.target.style.boxShadow="0 4px 22px rgba(255,60,154,0.55)";}}>
                  Play Again
                </button>
              </div>
            </div>
          )}
        </div>
      );
    }

    ReactDOM.createRoot(document.getElementById("root")).render(<Game/>);
